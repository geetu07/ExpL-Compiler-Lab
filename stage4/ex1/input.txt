begin
decl
    int n, i, j,k;
    int A[3][3],B[3][3],C[3][3];
enddecl

n=3;
i=0;
while(i < n) do
    j = 0;
    while(j < n) do
        read(A[i][j]);
        j = j + 1;
    endwhile
    i = i + 1;
endwhile
i=0;
while(i < n) do
    j = 0;
    while(j < n) do
        read(B[i][j]);
        j = j + 1;
    endwhile
    i = i + 1;
endwhile
i=0;
while(i < n) do
    j = 0;
    while(j < n) do
        C[i][j]=0;
        j = j + 1;
    endwhile
    i = i + 1;
endwhile
i = 0;
while(i < n) do
    j = 0;
    while(j < n) do
        k=0;
        while(k<n) do
            C[i][j]=C[i][j]+A[i][k]*B[k][j];
            k=k+1;
        endwhile
        j = j + 1;
    endwhile
    i = i + 1;
endwhile

i=0;
while(i < n) do
    j = 0;
    while(j < n) do
        write(C[i][j]);
        j = j + 1;
    endwhile
    i = i + 1;
endwhile

end;
